# Plano 4 — Arte: atlas animados, cenários, ícones, UI e validação visual (NarutoTxGame)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Preparar o jogo para receber arte de verdade sem nunca quebrar: lutadores animados por atlas (com fallback para o PNG estático atual e, depois, para o retângulo), movimento procedural enquanto não houver quadros desenhados, faíscas de impacto, cenários em camadas, ícones SVG, fontes e moldura de UI, e um pacote Playwright que fotografa as telas principais antes de qualquer push.

**Architecture:** Toda decisão testável fica em funções puras sem Phaser (src/scenes/fighterAnims.ts e `src/data/art.ts`), cobertas por Vitest no ambiente `node`. O Phaser só consome essas funções: o `BootScene` carrega o que o manifesto diz que existe e o `BattleScene` escolhe Sprite → Image → Rectangle conforme a textura carregou. A arte entra por scripts Python (Pillow) que geram public/assets/ a partir de assets/, como `tools/prepare-assets.py` já faz. O que a arena e o DOM mostram é conferido pelo pacote de validação visual (Task 1), que roda de novo ao fim de cada task.

**Tech Stack:** TypeScript 5 (strict), Vite 6, Phaser ~3.88.2 (Sprite, `load.atlas`, `anims.create`, partículas 3.60+), Vitest 5 (ambiente `node`; telas em happy-dom), Python 3 + Pillow 11, Playwright (devDependency nova, só para a ferramenta de screenshots), fonttools (só para gerar WOFF2, fora do repositório).

**Spec:** `docs/ARTE-PIPELINE.md` (formatos, convenção de nomes, rotas e decisões) + `docs/ASSETS.md` (inventário e prioridades P0/P1/P2). **Pré-requisito:** branch `feat/mvp` com `npm test`, `npm run build` e `npm run docs:check` verdes. **Antes da Task 1, a Task 0 precisa estar fechada** (decisões do Gabriel).

/** Funções puras do pacote de validação visual (tools/visual-check.ts). */
export interface PageProblem {
  kind: 'console' | 'pageerror' | 'asset';
  text: string;
}

/** Um arquivo de arte (ou do bundle, no dist/) que não carregou. */
export function isAssetFailure(url: string, status: number): boolean {
  if (status < 400) return false;
  try {
    return new URL(url).pathname.includes('/assets/');
  } catch {
    return false;
  }
}

/** O Chrome repete cada 404 como erro de Console; o 404 já entra pela resposta, com a URL. */
export function isIgnorableConsole(text: string): boolean {
  return text.startsWith('Failed to load resource');
}

const pad = (n: number) => String(n).padStart(2, '0');

export function stampDir(d: Date): string {
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}_${pad(d.getHours())}-${pad(d.getMinutes())}-${pad(d.getSeconds())}`;
}

export function shotName(n: number, scope: string, screen: string): string {
  return `${pad(n)}-${scope}-${screen}.png`;
}

export function report(problems: PageProblem[]): { ok: boolean; lines: string[] } {
  if (!problems.length) return { ok: true, lines: ['Nenhum problema: nenhum arquivo faltando e nenhum erro no Console.'] };
  const uniq = [...new Set(problems.map((p) => `[${p.kind}] ${p.text}`))];
  return { ok: false, lines: [`${uniq.length} problema(s):`, ...uniq.map((l) => `- ${l}`)] };
}
```

/**
 * Regras puras das animações dos lutadores (sem Phaser, testáveis no Node).
 * O BattleScene só aplica o que estas funções decidem.
 */
import { JUTSUS } from '../data/jutsus';
import type { GameEvent } from '../game/events';
import { DURATIONS } from './animationQueue';

/** Mesma ordem das linhas do atlas gerado por tools/pack-sprites.py. */
export const ANIM_NAMES = ['idle', 'attack', 'cast', 'hit', 'death', 'dodge'] as const;
export type AnimName = (typeof ANIM_NAMES)[number];

/** Quadros por segundo de cada animação (apresentação, não balanceamento). */
export const FRAME_RATE: Record<AnimName, number> = { idle: 6, attack: 14, cast: 12, hit: 12, death: 10, dodge: 14 };

/** Quadros sugeridos para quem desenha (docs/ASSETS.md §3.1). */
export const MAX_FRAMES: Record<AnimName, number> = { idle: 8, attack: 6, cast: 6, hit: 3, death: 6, dodge: 4 };

const d = (k: keyof typeof DURATIONS) => DURATIONS[k] ?? 0;

/**
 * Tempo máximo de cada animação, derivado dos passos da fila: o golpe precisa
 * terminar até o fim do dano que ele causa; o resto, dentro do próprio passo.
 */
export const ANIM_BUDGET_MS: Record<AnimName, number> = {
  idle: Number.POSITIVE_INFINITY,
  attack: d('actionChosen') + d('damage'),
  cast: d('jutsuUsed') + d('damage'),
  hit: d('damage'),
  death: d('defeated'),
  dodge: d('dodge'),
};

export function loops(anim: AnimName): boolean {
  return anim === 'idle';
}

export function holdsLastFrame(anim: AnimName): boolean {
  return anim === 'death';
}

const FRAME_RE = /^([a-z]+)_(\d{2,})$/;
const isAnim = (s: string): s is AnimName => (ANIM_NAMES as readonly string[]).includes(s);

/** 'idle_01', 'idle_00', 'attack_00' → { idle: ['idle_00', 'idle_01'], attack: ['attack_00'] }. */
export function groupFrames(frameNames: readonly string[]): Partial<Record<AnimName, string[]>> {
  const acc = new Map<AnimName, { n: number; name: string }[]>();
  for (const name of frameNames) {
    const m = FRAME_RE.exec(name);
    const anim = m?.[1];
    if (!m || !anim || !isAnim(anim)) continue;
    const list = acc.get(anim) ?? [];
    list.push({ n: Number(m[2]), name });
    acc.set(anim, list);
  }
  const out: Partial<Record<AnimName, string[]>> = {};
  for (const [anim, list] of acc) out[anim] = list.sort((a, b) => a.n - b.n).map((f) => f.name);
  return out;
}

export interface AnimCue {
  fighterId: string;
  anim: AnimName;
}

/** Que animação cada evento do motor pede, e para quem. */
export function cuesFor(e: GameEvent): AnimCue[] {
  switch (e.type) {
    case 'actionChosen':
      return e.action.type === 'attack' ? [{ fighterId: e.actorId, anim: 'attack' }] : [];
    case 'jutsuUsed':
      return [{ fighterId: e.actorId, anim: JUTSUS[e.jutsuId]?.category === 'taijutsu' ? 'attack' : 'cast' }];
    case 'damage':
    case 'dot':
      return [{ fighterId: e.targetId, anim: 'hit' }];
    case 'dodge':
    case 'miss':
      return [{ fighterId: e.targetId, anim: 'dodge' }];
    case 'defeated':
      return [{ fighterId: e.targetId, anim: 'death' }];
    default:
      return [];
  }
}

export function animDurationMs(frames: number, frameRate: number): number {
  return Math.round((frames / frameRate) * 1000);
}

/** Ritmo final: o padrão, acelerado se o atlas tiver quadros demais para o orçamento. */
export function frameRateFor(anim: AnimName, frames: number): number {
  if (loops(anim)) return FRAME_RATE[anim];
  return Math.max(FRAME_RATE[anim], Math.ceil((frames * 1000) / ANIM_BUDGET_MS[anim]));
}

/** Ordem do fallback: atlas animado, PNG estático, retângulo placeholder. */
export function bodyKind(has: { atlas: boolean; image: boolean }): 'sprite' | 'image' | 'rect' {
  if (has.atlas) return 'sprite';
  return has.image ? 'image' : 'rect';
}

/** `?boneco` na URL troca todos os lutadores pelo boneco de teste (conferir o pipeline sem arte). */
export function wantsDummyArt(search: string): boolean {
  return new URLSearchParams(search).has('boneco');
}

"""
Empacota quadros PNG soltos num atlas do Phaser (PNG + JSON Hash).
  python3 tools/pack-sprites.py                → todos os personagens de assets/sprites/
  python3 tools/pack-sprites.py player_semclan → só os nomeados
  python3 tools/pack-sprites.py --dummy        → gera antes o boneco de teste
Entrada: assets/sprites/<arte>/<animação>/<NN>.png, todos do mesmo tamanho, olhando para a DIREITA.
Saída:   public/assets/sprites/<arte>.png e <arte>.json, quadros '<animação>_<NN>'.
Uma linha por animação, na ordem de ANIMS (= ANIM_NAMES em src/scenes/fighterAnims.ts). Requer Pillow.
"""
import argparse
import json
from pathlib import Path

from PIL import Image, ImageDraw

ROOT = Path(__file__).resolve().parent.parent
SRC = ROOT / 'assets' / 'sprites'
DST = ROOT / 'public' / 'assets' / 'sprites'
ANIMS = ['idle', 'attack', 'cast', 'hit', 'death', 'dodge']
MAX_SIDE = 2048
DUMMY = 'boneco_teste'


def load_rows(art_dir: Path):
    rows = []
    for anim in ANIMS:
        folder = art_dir / anim
        files = sorted((p for p in folder.glob('*.png') if p.stem.isdigit()), key=lambda p: int(p.stem)) if folder.is_dir() else []
        if files:
            rows.append((anim, [Image.open(p).convert('RGBA') for p in files]))
    if not any(anim == 'idle' for anim, _ in rows):
        raise SystemExit(f'{art_dir.name}: falta a animação idle em {art_dir / "idle"}')
    return rows


def pack(art_dir: Path) -> None:
    rows = load_rows(art_dir)
    w, h = rows[0][1][0].size
    for anim, images in rows:
        for i, im in enumerate(images):
            if im.size != (w, h):
                raise SystemExit(f'{art_dir.name}/{anim}/{i:02d}.png mede {im.size}; todos os quadros precisam medir {(w, h)}')
    cols = max(len(images) for _, images in rows)
    size = (cols * w, len(rows) * h)
    if max(size) > MAX_SIDE:
        raise SystemExit(f'{art_dir.name}: folha de {size} passa de {MAX_SIDE} px; reduza quadros ou o tamanho do quadro')
    sheet = Image.new('RGBA', size, (0, 0, 0, 0))
    frames = {}
    for r, (anim, images) in enumerate(rows):
        for c, im in enumerate(images):
            x, y = c * w, r * h
            sheet.paste(im, (x, y))
            frames[f'{anim}_{c:02d}'] = {
                'frame': {'x': x, 'y': y, 'w': w, 'h': h}, 'rotated': False, 'trimmed': False,
                'spriteSourceSize': {'x': 0, 'y': 0, 'w': w, 'h': h}, 'sourceSize': {'w': w, 'h': h},
            }
    DST.mkdir(parents=True, exist_ok=True)
    png = DST / f'{art_dir.name}.png'
    sheet.save(png, optimize=True)
    atlas = {'frames': frames, 'meta': {'image': png.name, 'size': {'w': size[0], 'h': size[1]}, 'scale': '1'}}
    (DST / f'{art_dir.name}.json').write_text(json.dumps(atlas, indent=1) + '\n', encoding='utf-8')
    print(f'{png.relative_to(ROOT)}  {size[0]}x{size[1]}  {len(frames)} quadros  {png.stat().st_size // 1024} KB')


# Boneco de teste: cor por animação + número do quadro, para ver na arena qual animação toca.
DUMMY_FRAMES = {'idle': 4, 'attack': 4, 'cast': 4, 'hit': 2, 'death': 4, 'dodge': 3}
DUMMY_COLOR = {'idle': (90, 140, 255), 'attack': (240, 138, 36), 'cast': (185, 140, 232),
               'hit': (224, 82, 74), 'death': (120, 120, 120), 'dodge': (76, 175, 106)}


def make_dummy() -> None:
    for anim, count in DUMMY_FRAMES.items():
        folder = SRC / DUMMY / anim
        folder.mkdir(parents=True, exist_ok=True)
        for i in range(count):
            im = Image.new('RGBA', (96, 96), (0, 0, 0, 0))
            d = ImageDraw.Draw(im)
            bob = (i % 2) * 2

  private makeBody(art: CharacterArt, side: 'player' | 'enemy', color: number): Pick<Fighter, 'body' | 'height' | 'anims' | 'baseScale'> {
    const atlas = atlasName(art, this.dummy);
    const atlasKey = atlas ? atlasTextureKey(atlas) : null;
    const kind = bodyKind({ atlas: atlasKey !== null && this.textures.exists(atlasKey), image: this.textures.exists(art) });
    if (kind === 'rect') {
      return { body: this.add.rectangle(0, 0, 110, 170, color).setStrokeStyle(3, 0xffffff, 0.6), height: 170, anims: {}, baseScale: 1 };

      sprite.setFlipX(side === 'enemy');

  /** Cria (uma vez por textura) as animações do Phaser a partir dos nomes dos quadros do atlas. */
  private ensureAnims(textureKey: string): { anims: Partial<Record<AnimName, string>>; firstFrame: string | undefined } {
    const groups = groupFrames(this.textures.get(textureKey).getFrameNames());
    const anims: Partial<Record<AnimName, string>> = {};
    for (const [anim, frames] of Object.entries(groups) as [AnimName, string[]][]) {
      const key = `${textureKey}:${anim}`;
      if (!this.anims.exists(key)) {
        this.anims.create({ key, frames: frames.map((frame) => ({ key: textureKey, frame })), frameRate: frameRateFor(anim, frames.length), repeat: loops(anim) ? -1 : 0 });

      anims[anim] = key;
    }

  /** Toca a animação pedida: quadros do atlas se existirem, senão movimento por tween. */
  private cue(c: AnimCue): void {
    const f = this.fighters.get(c.fighterId);
    if (!f || f.dead) return;
    if (c.anim === 'death') f.dead = true;
    const body = f.body;
    if (body instanceof Phaser.GameObjects.Rectangle) return;
    const key = f.anims[c.anim];
    if (body instanceof Phaser.GameObjects.Sprite && key) {
      body.play(key);
      const idle = f.anims.idle;
      if (idle && !loops(c.anim) && !holdsLastFrame(c.anim)) {
        body.once(Phaser.Animations.Events.ANIMATION_COMPLETE, () => {
          if (!f.dead) body.play(idle);
        });
      }
      return;
    }
    this.procedural(f, body, c.anim);
  }

  /** Movimento por tween para PNG estático ou atlas sem aquela animação. */
  private procedural(f: Fighter, body: Phaser.GameObjects.Sprite | Phaser.GameObjects.Image, anim: AnimName): void {
    const s = f.baseScale;
    const towardCenter = f.baseX < 640 ? 1 : -1;
    switch (anim) {
      case 'hit':
        this.tweens.add({ targets: body, scaleX: s * (1 + HIT_SQUASH), scaleY: s * (1 - HIT_SQUASH), duration: HIT_MS, yoyo: true, onComplete: () => body.setScale(s) });
        return;
      case 'dodge':
        this.tweens.add({ targets: f.container, x: f.baseX - DODGE_PX * towardCenter, duration: DODGE_MS, yoyo: true, onComplete: () => f.container.setX(f.baseX) });
        return;
      case 'death':
        this.tweens.killTweensOf(body);
        this.tweens.add({ targets: body, angle: -FALL_ANGLE * towardCenter, y: FALL_PX, duration: FALL_MS, ease: 'Quad.easeIn' });
        return;
      default:
        // idle é a respiração (resetFighters); attack e cast já são o lunge do contêiner.
        return;
    }
  }

export function battleBackground(nodeIndex: number | null, saga: SagaDef = SAGA_ONDA): BattleBackground {
  const node = nodeIndex === null ? undefined : saga.nodes[nodeIndex];
  if (!node) return 'estrada_floresta';

  return firstBoss && node.index > firstBoss.index ? 'costa_nevoa' : 'estrada_floresta';
}

"""
Cenários procedurais de batalha (Rota B de docs/ARTE-PIPELINE.md). Primeira versão,
para trocar depois por arte desenhada com o MESMO nome de arquivo.
  python3 tools/make-backgrounds.py
Escreve public/assets/cenarios/<cenario>_{ceu,meio,chao}.webp em 1280 × 720.
A linha do chão fica em y = 510, onde pisam os lutadores (FIGHTER_Y + SPRITE_HEIGHT / 2
em src/scenes/BattleScene.ts). Requer Pillow.
"""
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFilter

ROOT = Path(__file__).resolve().parent.parent
DST = ROOT / 'public' / 'assets' / 'cenarios'
W, H, GROUND_Y = 1280, 720, 510

# (céu topo, céu base), silhueta do meio, chão. Tons escuros: o HUD fica por cima.
PALETTES = {
    'estrada_floresta': ((0x1b2a3a, 0x3d5a4a), 0x1a2b20, 0x2b2418),
    'costa_nevoa': ((0x1e2633, 0x55657a), 0x2a3340, 0x2e3036),
    'lago_nevoa': ((0x1a2230, 0x4a5d70), 0x23303c, 0x1f3442),
    'ponte': ((0x20202c, 0x5a5560), 0x2b2620, 0x3a2e22),
}


def rgb(c):
    return ((c >> 16) & 255, (c >> 8) & 255, c & 255)


def sky(top, bottom):
    im = Image.new('RGB', (W, H))
    d = ImageDraw.Draw(im)
    t, b = rgb(top), rgb(bottom)
    for y in range(H):
        k = y / (H - 1)
        d.line([(0, y), (W, y)], fill=tuple(round(t[i] + (b[i] - t[i]) * k) for i in range(3)))
    return im


def middle(name, color, rnd):
    im = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    c = rgb(color) + (255,)
    if name == 'estrada_floresta':
        for x in range(-40, W + 40, 70):
            h = rnd.randint(180, 300)
            d.polygon([(x, GROUND_Y), (x + 45, GROUND_Y - h), (x + 90, GROUND_Y)], fill=c)
    elif name in ('costa_nevoa', 'lago_nevoa'):
        pts = [(0, GROUND_Y)] + [(x, GROUND_Y - rnd.randint(60, 160)) for x in range(0, W + 80, 80)] + [(W, GROUND_Y)]
        d.polygon(pts, fill=c)
        im = im.filter(ImageFilter.GaussianBlur(3))  # névoa
    else:  # ponte: pilares e tabuleiro
        for x in range(60, W, 160):
            d.rectangle([x, GROUND_Y - 220, x + 18, GROUND_Y], fill=c)
        d.rectangle([0, GROUND_Y - 230, W, GROUND_Y - 214], fill=c)
    return im


def ground(color):
    im = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    d.rectangle([0, GROUND_Y, W, H], fill=rgb(color) + (255,))
    d.line([(0, GROUND_Y), (W, GROUND_Y)], fill=(255, 255, 255, 60), width=2)  # linha dos pés
    return im


def main():
    DST.mkdir(parents=True, exist_ok=True)
    for name, (sky_c, mid_c, ground_c) in PALETTES.items():
        rnd = random.Random(name)  # mesma saída a cada execução
        layers = {'ceu': sky(*sky_c), 'meio': middle(name, mid_c, rnd), 'chao': ground(ground_c)}
        for layer, im in layers.items():
            out = DST / f'{name}_{layer}.webp'
            im.save(out, 'WEBP', quality=80, method=6)
            print(f'{out.relative_to(ROOT)}  {out.stat().st_size // 1024} KB')


if __name__ == '__main__':
    main()
```
